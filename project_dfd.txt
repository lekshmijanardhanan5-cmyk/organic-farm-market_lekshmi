%% Project DFD (Data Flow Diagram) - MERN "Organic Farm Market"
%% Mermaid diagrams are provided for easy reuse.
%% ============================================================

%% =========================
%% DFD Level 0 (Entire Project)
%% =========================
```mermaid
flowchart TD
  %% External Actors
  Customer["Customer Browser / Mobile"]:::ext
  Farmer["Farmer Browser / Mobile"]:::ext
  Admin["Admin Browser / Mobile"]:::ext

  %% Frontend
  UI["Frontend (React + Router + AuthContext)"]:::frontend

  %% Backend API (Express)
  API["Backend API (Express Routes + Middleware)"]:::backend

  %% Data Store
  DB["MongoDB (User, Product, Order, Transaction)"]:::db

  %% Real-time updates
  SSE["SSE Order Stream (/api/orders/subscribe/:orderId)"]:::realtime
  Events["orderEvents (EventEmitter)"]:::internal

  %% Payment Simulation
  Gateway["Simulated Checkout Gateway (/checkout/initiate, /checkout/confirm)"]:::internal

  %% Flows: Auth
  Customer --> UI
  Farmer --> UI
  Admin --> UI
  UI -->|JWT login/register/profile| API
  API --> DB

  %% Flows: Public/Role-based Product browsing
  UI -->|GET /api/products| API
  UI -->|GET /api/products/:id| API
  API --> DB

  %% Flows: Reviews (customer)
  UI -->|GET /api/reviews/product/:id| API
  UI -->|GET /api/reviews/can-review/:productId| API
  UI -->|POST /api/reviews| API
  API --> DB

  %% Flows: Orders (customer checkout)
  UI -->|POST /api/orders/checkout/initiate| API
  UI -->|POST /api/orders/checkout/confirm| API
  UI -->|POST /api/orders (legacy)| API
  API --> Gateway
  API --> DB

  %% Flows: Order status updates (farmer/admin via dashboard)
  UI -->|PUT /api/orders/:id/status| API
  API --> Events

  %% Flows: Real-time push back to customers/farmers/admins
  API --> SSE
  Events --> SSE
  SSE --> UI

  %% Flows: Orders & Transactions views by role
  UI -->|GET /api/orders/user| API
  UI -->|GET /api/orders/farmer| API
  UI -->|GET /api/admin/orders| API
  UI -->|GET /api/transactions/customer| API
  UI -->|GET /api/transactions/farmer| API
  UI -->|GET /api/transactions/admin| API
  API --> DB

  %% Admin oversight
  UI -->|PUT /api/admin/users/:id/approve| API
  UI -->|PUT /api/admin/users/:id/block| API
  UI -->|DELETE /api/admin/users/:id| API
  UI -->|Admin products/orders/users/stats CRUD| API
  API --> DB

  classDef ext fill:#fff,stroke:#333,stroke-width:1px;
  classDef frontend fill:#e6f7ff,stroke:#006d75,stroke-width:1px;
  classDef backend fill:#fff7e6,stroke:#ad6b00,stroke-width:1px;
  classDef db fill:#f5f5f5,stroke:#555,stroke-width:1px;
  classDef realtime fill:#ffe6e6,stroke:#cc0000,stroke-width:1px;
  classDef internal fill:#f0fff4,stroke:#2b8a3e,stroke-width:1px;
```

%% =========================
%% DFD Level 1 (Modules & Major Data Stores)
%% =========================
```mermaid
flowchart TD
  %% Actors
  Customer["Customer"]:::ext
  Farmer["Farmer"]:::ext
  Admin["Admin"]:::ext

  %% Frontend modules
  AuthUI["Frontend Auth UI (login/register/profile sync)"]:::frontend
  ProdUI["Frontend Product UI (browse, details, reviews)"]:::frontend
  CustDash["Customer Dashboard UI (orders, transactions, order status)"]:::frontend
  FarmerDash["Farmer Dashboard UI (fulfillment, products/orders)"]:::frontend
  AdminDash["Admin Dashboard UI (approval/blocking, manage all)"]:::frontend

  %% Backend route modules
  AuthAPI["Auth/Profile API (/api/auth)"]:::backend
  ProductAPI["Products API (/api/products)"]:::backend
  ReviewAPI["Reviews API (/api/reviews)"]:::backend
  OrderAPI["Orders API (/api/orders) incl. checkout & SSE"]:::backend
  TransactionAPI["Transactions API (/api/transactions)"]:::backend
  AdminAPI["Admin API (/api/admin)"]:::backend
  FarmerAPI["Farmer API (/api/farmer)"]:::backend
  CustomerAPI["Customer API (/api/customer)"]:::backend

  %% Simulated checkout gateway (internal)
  Gateway["Simulated Checkout Gateway"]:::internal

  %% Real-time
  SSE["SSE Stream (/api/orders/subscribe/:orderId)"]:::realtime
  Events["orderEvents EventEmitter"]:::internal

  %% Data Store
  DB["MongoDB Data Store"]:::db

  %% Frontend -> Backend
  Customer --> AuthUI
  Customer --> ProdUI
  Customer --> CustDash
  Farmer --> FarmerDash
  Admin --> AdminDash

  AuthUI -->|JWT| AuthAPI
  ProdUI --> ProductAPI
  ProdUI --> ReviewAPI

  CustDash -->|Order placement & history| OrderAPI
  CustDash -->|Transactions history| TransactionAPI
  CustDash -->|Subscribe SSE| SSE

  FarmerDash --> FarmerAPI
  FarmerDash -->|Orders list + status update| OrderAPI
  FarmerDash -->|Transactions history| TransactionAPI
  FarmerDash --> SSE

  AdminDash --> AdminAPI
  AdminDash -->|All orders/products/users/stats| AdminAPI
  AdminDash -->|Transactions history| TransactionAPI

  %% Backend -> Internal modules
  OrderAPI --> Gateway
  OrderAPI --> Events

  %% Backend -> Data Store
  AuthAPI --> DB
  ProductAPI --> DB
  ReviewAPI --> DB
  OrderAPI --> DB
  TransactionAPI --> DB
  AdminAPI --> DB
  FarmerAPI --> DB
  CustomerAPI --> DB

  %% Real-time SSE updates
  Events --> SSE
  SSE --> CustDash
  SSE --> FarmerDash
  SSE --> AdminDash

  classDef ext fill:#fff,stroke:#333,stroke-width:1px;
  classDef frontend fill:#e6f7ff,stroke:#006d75,stroke-width:1px;
  classDef backend fill:#fff7e6,stroke:#ad6b00,stroke-width:1px;
  classDef db fill:#f5f5f5,stroke:#555,stroke-width:1px;
  classDef realtime fill:#ffe6e6,stroke:#cc0000,stroke-width:1px;
  classDef internal fill:#f0fff4,stroke:#2b8a3e,stroke-width:1px;
```

%% =========================
%% DFD Level 1.1 (Customer Modules)
%% =========================
```mermaid
flowchart TD
  Customer["Customer"]:::ext
  UI["Customer-facing UI (Products + Details + Dashboard)"]:::frontend
  API["Backend Orders/Products/Reviews/Transactions APIs"]:::backend
  DB["MongoDB"]:::db
  SSE["SSE: /api/orders/subscribe/:orderId"]:::realtime
  Events["orderEvents"]:::internal
  Gateway["Simulated Checkout Gateway"]:::internal

  %% Browse & Details
  UI -->|GET /api/products| API
  UI -->|GET /api/products/:id| API
  API --> DB

  %% Reviews (eligible after delivered)
  UI -->|GET /api/reviews/product/:id| API
  UI -->|GET /api/reviews/can-review/:productId| API
  UI -->|POST /api/reviews| API
  API --> DB

  %% Checkout (COD / UPI / CARD)
  UI -->|POST /api/orders/checkout/initiate| API
  UI -->|POST /api/orders/checkout/confirm| API
  UI -->|POST /api/orders (legacy)| API
  API --> Gateway
  API --> DB

  %% Persisted Outcomes
  DB -->|Order (paymentMethod, paymentStatus, optional upiId/cardLast4)| API
  DB -->|Transaction ledger for the order| API

  %% Subscribe for status updates (Pending -> Accepted -> Packed -> Delivered)
  UI -->|SSE subscribe with JWT token (header or ?token=)| SSE
  Events --> SSE
  SSE --> UI

  %% Customer history views
  UI -->|GET /api/orders/user?status=...| API
  UI -->|GET /api/transactions/customer| API
  API --> DB

  classDef ext fill:#fff,stroke:#333,stroke-width:1px;
  classDef frontend fill:#e6f7ff,stroke:#006d75,stroke-width:1px;
  classDef backend fill:#fff7e6,stroke:#ad6b00,stroke-width:1px;
  classDef db fill:#f5f5f5,stroke:#555,stroke-width:1px;
  classDef realtime fill:#ffe6e6,stroke:#cc0000,stroke-width:1px;
  classDef internal fill:#f0fff4,stroke:#2b8a3e,stroke-width:1px;
```

%% DFD Level 1.2 (Farmer Modules)
%% =========================
```mermaid
flowchart TD
  Farmer["Farmer"]:::ext
  UI["Farmer-facing UI (Dashboard + Products + Fulfillment)"]:::frontend
  API["Backend Farmer/Orders/Products/Transactions APIs"]:::backend
  DB["MongoDB"]:::db
  SSE["SSE: /api/orders/subscribe/:orderId"]:::realtime
  Events["orderEvents"]:::internal

  %% View and manage farmer products
  UI -->|GET /api/farmer/products| API
  UI -->|POST /api/products (create owned products)| API
  UI -->|PUT /api/products/:id (update owned product)| API
  UI -->|DELETE /api/products/:id (delete owned product)| API
  API --> DB

  %% Farmer orders for own products
  UI -->|GET /api/orders/farmer?status=...| API
  API --> DB

  %% Fulfillment updates (status progression)
  UI -->|PUT /api/orders/:id/status (Accept/Packed/Delivered)| API
  API --> Events

  %% Farmer transaction history
  UI -->|GET /api/transactions/farmer| API
  API --> DB

  %% Optional real-time updates
  UI -->|SSE subscribe (authorized for orders containing farmer's products)| SSE
  Events --> SSE
  SSE --> UI
```
